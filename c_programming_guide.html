Introduction to C

C is the mother of all modern programming languages. Created in 1972 by Dennis Ritchie at Bell Labs, it powers operating systems, embedded systems, and everything in between.

🌍
What is C?

C is a general-purpose, compiled, procedural programming language. It gives you direct access to memory and hardware — making it extremely fast and powerful.

Almost every modern language (C++, Java, Python, JavaScript) was influenced by C. If you understand C, you understand how computers truly work.

Compiled Low-level control Super fast Portable

Your First C Program — Hello World

Every C program starts with a main() function. This is the entry point — where execution begins.

C — hello.c
#include <stdio.h>    // Standard input/output library

int main() {
    printf("Hello, World!\n");  // Print to screen
    return 0;                  // Tell OS: program ran fine
}
▶ Output
Hello, World!
💡 Line by Line:
#include <stdio.h> — Includes the standard I/O library (gives us printf)
int main() — Main function, returns an integer
printf() — Prints text to the screen
\n — Newline character (go to next line)
return 0 — 0 means success to the operating system

Constants — Values That Never Change
C — constants.c
#include <stdio.h>

// Method 1: #define (macro constant)
#define MAX_SIZE   100
#define PI        3.14159265
#define GREETING  "Hello!"

int main() {
    // Method 2: const keyword
    const int DAYS_IN_WEEK = 7;
    const float GRAVITY = 9.81;
    
    printf("Max Size: %d\n", MAX_SIZE);
    printf("PI: %.8f\n", PI);
    printf("Days in Week: %d\n", DAYS_IN_WEEK);
    printf("Gravity: %.2f m/s²\n", GRAVITY);

    return 0;
}

Max Size: 100 PI: 3.14159265 Days in Week: 7 Gravity: 9.81 m/s² Hello!
⚠️ #define vs const: #define is handled by the preprocessor (no type, no memory). const is type-safe and checked by the compiler. Prefer const in modern C.

Type Casting
C — typecast.c
#include <stdio.h>

int main() {
    int a = 7, b = 2;
    
    // Without casting — integer division!
    float result1 = a / b;
    printf("Without cast: %f\n", result1);   // 3.000000
    
    // With explicit casting — correct!
    float result2 = (float)a / b;
    printf("With cast: %f\n", result2);      // 3.500000
    
    // int → char
    int num = 65;
    char ch = (char)num;
    printf("int 65 as char: %c\n", ch);      // 'A'
    
    // float → int (loses decimal)
    float pi = 3.99;
    int piInt = (int)pi;
    printf("3.99 as int: %d\n", piInt);      // 3 (truncated!)
    
    return 0;
}
▶ Output
Without cast: 3.000000 With cast: 3.500000 int 65 as char: A 3.99 as int: 3
⚠️ Watch Out: Converting float→int truncates (not rounds). 3.99 becomes 3, not 4!

Relational & Logical Operators
C — logical.c
#include <stdio.h>

int main() {
    int a = 10, b = 20, c = 10;
    
    // Relational (return 1=true, 0=false)
    printf("a == c : %d\n", a == c);   // 1
    printf("a != b : %d\n", a != b);   // 1
    printf("a < b  : %d\n", a < b);    // 1
    printf("a > b  : %d\n", a > b);    // 0
    printf("a <= c : %d\n", a <= c);  // 1
    
    // Logical: && (AND), || (OR), ! (NOT)
    printf("a<b && a==c : %d\n", (a<b) && (a==c)); // 1
    printf("a>b || a==c : %d\n", (a>b) || (a==c)); // 1
    printf("!(a == b)   : %d\n", !(a==b));           // 1
    
    return 0;
}
▶ Output
a == c : 1 a != b : 1 a < b : 1 a > b : 0 a <= c : 1 ab || a==c : 1 !(a == b) : 1

Recursion — Function Calling Itself
C — recursion.c
#include <stdio.h>

// Factorial: n! = n * (n-1) * ... * 1
int factorial(int n) {
    if (n == 0 || n == 1)
        return 1;     // Base case
    return n * factorial(n - 1);  // Recursive call
}

// Fibonacci: 0,1,1,2,3,5,8,13...
int fibonacci(int n) {
    if (n <= 1) return n;
    return fibonacci(n-1) + fibonacci(n-2);
}

int main() {
    printf("Factorials:\n");
    for (int i = 0; i <= 7; i++)
        printf("%d! = %d\n", i, factorial(i));
    
    printf("\nFibonacci (first 10):\n");
    for (int i = 0; i < 10; i++)
        printf("%d ", fibonacci(i));
    
    return 0;
}

Arrays

An array is a collection of elements of the same type stored in contiguous memory. Arrays are the backbone of C programming.

📋
1D Arrays — Declare, Access, Modify
C — arrays.c
#include <stdio.h>

int main() {
    // Declare and initialize
    int nums[5] = {10, 20, 30, 40, 50};
    
    // Access by index (starts at 0!)
    printf("First: %d\n", nums[0]);    // 10
    printf("Last: %d\n",  nums[4]);    // 50
    
    // Modify
    nums[2] = 999;
    
    // Print all with loop
    printf("Array: ");
    for (int i = 0; i < 5; i++)
        printf("%d ", nums[i]);
    
    // Find max and min
    int max = nums[0], min = nums[0];
    for (int i = 1; i < 5; i++) {
        if (nums[i] > max) max = nums[i];
        if (nums[i] < min) min = nums[i];
    }
    printf("\nMax: %d, Min: %d\n", max, min);
    
    return 0;
}
▶ Output
First: 10 Last: 50 Array: 10 20 999 40 50 Max: 999, Min: 10

Strings

In C, a string is an array of characters ending with a null terminator '\0'. The string.h library provides powerful string functions.

💬
String Basics & Functions
C — strings.c
#include <stdio.h>
#include <string.h>    // For string functions
#include <ctype.h>     // For toupper(), tolower()

int main() {
    char str1[50] = "Hello";
    char str2[50] = "World";
    char str3[100];
    
    // strlen — length (not counting \0)
    printf("Length of '%s': %zu\n", str1, strlen(str1));
    
    // strcpy — copy string
    strcpy(str3, str1);
    printf("Copied: %s\n", str3);
    
    // strcat — concatenate (join)
    strcat(str1, " ");
    strcat(str1, str2);
    printf("Joined: %s\n", str1);
    
    // strcmp — compare (0 = equal)
    char a[] = "apple", b[] = "apple";
    printf("strcmp: %d\n", strcmp(a, b));   // 0
    
    // strchr — find character
    char sentence[] = "I love C programming";
    char* pos = strchr(sentence, 'C');
    printf("Found 'C' at: %s\n", pos);
    
    // Convert to uppercase
    char word[] = "hello";
    for (int i = 0; word[i]; i++)
        word[i] = toupper(word[i]);
    printf("Uppercase: %s\n", word);
    
    return 0;
}
▶ Output
Length of 'Hello': 5 Copied: Hello Joined: Hello World strcmp: 0 Found 'C' at: C programming Uppercase: HELLO
Pointers

Pointers are what make C special and powerful. A pointer stores the MEMORY ADDRESS of another variable. This is C's superpower — direct memory manipulation.

👉
Pointer Basics — Address & Dereference
C — pointers.c
#include <stdio.h>

int main() {
    int x = 42;          // regular variable
    int* ptr = &x;        // ptr stores ADDRESS of x
    
    printf("Value of x     : %d\n", x);
    printf("Address of x   : %p\n", &x);
    printf("ptr holds      : %p\n", ptr);
    printf("Value via ptr  : %d\n", *ptr);   // dereference
    
    // Change value through pointer!
    *ptr = 100;
    printf("x is now       : %d\n", x);     // 100
    
    // Pointer arithmetic
    int arr[] = {10, 20, 30, 40};
    int* p = arr;         // array name = pointer to first element
    printf("\nArray via pointer:\n");
    for (int i = 0; i < 4; i++)
        printf("*(p+%d) = %d\n", i, *(p+i));
    
    return 0;
}

🔄
Pointers in Functions — Swap Example
C — swap.c
#include <stdio.h>

// Pass by VALUE — does NOT change original
void swapBad(int a, int b) {
    int temp = a; a = b; b = temp;
    // Only local copies change!
}

// Pass by POINTER — changes original!
void swapGood(int* a, int* b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main() {
    int x = 5, y = 10;
    
    printf("Before: x=%d, y=%d\n", x, y);
    
    swapBad(x, y);
    printf("After swapBad: x=%d, y=%d\n", x, y); // unchanged!
    
    swapGood(&x, &y);
    printf("After swapGood: x=%d, y=%d\n", x, y); // swapped!
    
    return 0;
}
▶ Output
Before: x=5, y=10 After swapBad: x=5, y=10 After swapGood: x=10, y=5
💡 Key Insight: In C, all function arguments are passed by VALUE (copies). To modify the original, pass a pointer (address) using &.

File I/O

C can read and write files using the FILE* type from stdio.h. This lets your programs store data permanently on disk.

📁
Writing & Reading Files
C — fileio.c
#include <stdio.h>
#include <stdlib.h>

int main() {
    FILE* fp;
    
    // ===== WRITE to file =====
    fp = fopen("data.txt", "w");  // "w" = write mode
    if (fp == NULL) {
        printf("Error creating file!\n");
        return 1;
    }
    fprintf(fp, "Name: Pavel\n");
    fprintf(fp, "Age: 18\n");
    fprintf(fp, "GPA: 3.85\n");
    fclose(fp);  // ALWAYS close the file!
    printf("File written successfully!\n");
    
    // ===== READ from file =====
    fp = fopen("data.txt", "r");  // "r" = read mode
    char line[100];
    printf("\n--- File Contents ---\n");
    while (fgets(line, sizeof(line), fp) != NULL)
        printf("%s", line);
    fclose(fp);
    
    return 0;
}
▶ Output
File written successfully! --- File Contents --- Name: Pavel Age: 18 GPA: 3.85
💡 File Modes: "w" write (creates/overwrites), "r" read, "a" append, "r+" read+write, "w+" write+read, "rb"/"wb" binary modes

Dynamic Memory — malloc, calloc, free
C — dynamic_memory.c
#include <stdio.h>
#include <stdlib.h>

int main() {
    int n;
    printf("How many numbers? ");
    scanf("%d", &n);
    
    // malloc — allocate n*4 bytes
    int* arr = (int*) malloc(n * sizeof(int));
    
    if (arr == NULL) {
        printf("Memory allocation failed!\n");
        return 1;
    }
    
    // Use the dynamic array
    for (int i = 0; i < n; i++) {
        arr[i] = (i+1) * 10;
    }
    
    printf("Dynamic array: ");
    for (int i = 0; i < n; i++)
        printf("%d ", arr[i]);
    
    // MUST free to prevent memory leaks!
    free(arr);
    printf("\nMemory freed successfully!\n");
    
    // calloc — allocates AND initializes to 0
    int* arr2 = (int*) calloc(5, sizeof(int));
    printf("calloc array (all zeros): ");
    for (int i = 0; i < 5; i++)
        printf("%d ", arr2[i]);
    free(arr2);
    
    return 0;
}
▶ Output (input: 4)
How many numbers? 4 Dynamic array: 10 20 30 40 Memory freed successfully! calloc array (all zeros): 0 0 0 0 0
⚠️ Memory Leak: Every malloc() or calloc() MUST have a matching free(). Forgetting to free causes memory leaks — a critical bug in production code!

Complete Program: Student Grade System

A full program combining everything — arrays, structs, functions, loops, conditions, and file output.

C — grade_system.c (Complete Program)
#include <stdio.h>
#include <string.h>

typedef struct {
    char name[30];
    int math, science, english;
    float avg;
    char grade;
} Student;

char getGrade(float avg) {
    if (avg >= 90) return 'A';
    if (avg >= 80) return 'B';
    if (avg >= 70) return 'C';
    if (avg >= 60) return 'D';
    return 'F';
}

void printReport(Student s) {
    printf("| %-12s | %5d | %7d | %7d | %5.1f | %5c |\n",
        s.name, s.math, s.science, s.english, s.avg, s.grade);
}

int main() {
    Student students[4] = {
        {"Pavel",   92, 88, 95},
        {"Rahim",   75, 82, 70},
        {"Karim",   60, 55, 65},
        {"Sadia",   98, 96, 99}
    };
    
    // Calculate avg and grade
    for (int i = 0; i < 4; i++) {
        students[i].avg = (students[i].math +
                           students[i].science +
                           students[i].english) / 3.0;
        students[i].grade = getGrade(students[i].avg);
    }
    
    // Print report
    printf("+==============+=======+=========+=========+=======+=======+\n");
    printf("| Name         |  Math | Science | English |   Avg | Grade |\n");
    printf("+==============+=======+=========+=========+=======+=======+\n");
    for (int i = 0; i < 4; i++)
        printReport(students[i]);
    printf("+==============+=======+=========+=========+=======+=======+\n");
    
    // Find topper
    int topIdx = 0;
    for (int i = 1; i < 4; i++)
        if (students[i].avg > students[topIdx].avg)
            topIdx = i;
    printf("\n★ Topper: %s (Avg: %.1f)\n",
           students[topIdx].name, students[topIdx].avg);
    
    return 0;
}
